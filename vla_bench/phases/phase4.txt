x_et_xnnpack_8da8w        | [ -f models/et/smol-xnnpack-8da8w/model.pte ] || .venvs/executorch/bin/python engines/export_executorch.py --recipe xnnpack --dtype float32 --qlinear 8da8w --out models/et/smol-xnnpack-8da8w
x_et_xnnpack_8da4w        | [ -f models/et/smol-xnnpack-8da4w/model.pte ] || .venvs/executorch/bin/python engines/export_executorch.py --recipe xnnpack --dtype float32 --qlinear 8da4w --qlinear-group-size 32 --out models/et/smol-xnnpack-8da4w

x_et_cuda_int4            | [ -f models/et/smol-cuda-int4/model.pte ] || CUDA_HOME=/usr/local/cuda PATH=/usr/local/cuda/bin:$PATH .venvs/executorch/bin/python engines/export_executorch.py --recipe cuda --dtype bfloat16 --device cuda --qlinear 4w --qlinear-packing-format tile_packed_to_4d --out models/et/smol-cuda-int4
